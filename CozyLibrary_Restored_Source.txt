local function mergeOptions(options: any?): any
	local result = {}
	for key, value in pairs(DEFAULTS) do
		(result :: any)[key] = value
	end
	if options then
		for key, value in pairs(options) do
			(result :: any)[key] = value
		end
	end
	return result
end

local function playPurpleStarRemoval(player: Player)
	local character = player.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root or not root:IsA("BasePart") then return end
	local billboard = Instance.new("BillboardGui")
	billboard.Name = "CozyDuplicateCleanupStar"
	billboard.Adornee = root
	billboard.AlwaysOnTop = true
	billboard.Size = UDim2.fromOffset(76, 76)
	billboard.StudsOffset = Vector3.new(0, 1.5, 0)
	billboard.Parent = root
	local star = Instance.new("TextLabel")
	star.Size = UDim2.fromScale(1, 1)
	star.BackgroundTransparency = 1
	star.Text = "✦"
	star.TextColor3 = Color3.fromRGB(218, 164, 255)
	star.TextStrokeColor3 = Color3.fromRGB(104, 49, 158)
	star.TextStrokeTransparency = 0.22
	star.TextScaled = true
	star.Font = Enum.Font.GothamBlack
	star.Parent = billboard
	task.spawn(function()
		for step = 0, 1, (1 / (120 * 1.2)) do
			if not billboard.Parent then return end
			local angle = step * math.pi * 4
			billboard.StudsOffset = Vector3.new(math.cos(angle) * 1.8, 1.5 + step * 7, math.sin(angle) * 1.8)
			star.Rotation = step * 720
			star.TextTransparency = step * 0.9
			task.wait(1 / 120)
		end
		billboard:Destroy()
	end)
end

local function cleanupDuplicateGuis(player: Player, parent: Instance, guiName: string)
	for _, child in ipairs(parent:GetChildren()) do
		if child:IsA("ScreenGui") and child.Name == guiName then
			playPurpleStarRemoval(player)
			local reporter = shared.CozyReportDuplicate
			if type(reporter) == "function" then pcall(reporter, guiName) end
			child:Destroy()
		end
	end
end

local function cloneSafeViewportModel(source: Instance?, world: WorldModel, player: Player): Model?
	local candidate = source
	if not candidate then candidate = player.Character end
	if not candidate or not candidate:IsA("Model") then return nil end
	local ok, clone = pcall(function() return candidate:Clone() end)
	if not ok or not clone or not clone:IsA("Model") then return nil end
	for _, object in ipairs(clone:GetDescendants()) do
		if object:IsA("Script") or object:IsA("LocalScript") or object:IsA("ModuleScript") then object:Destroy() end
		if object:IsA("BasePart") then object.Anchored = true; object.CanCollide = false; object.CastShadow = false end
	end
	clone:PivotTo(CFrame.new(0, -2.2, 0))
	clone.Parent = world
	return clone
end

local function collectClientVisibleRemotes(root: Instance): string
	local records = {}
	local count = 0
	for _, object in ipairs(root:GetDescendants()) do
		if object:IsA("RemoteEvent") or object:IsA("RemoteFunction") or object:IsA("UnreliableRemoteEvent") then
			count += 1
			if count <= 80 then table.insert(records, string.format("- %s | %s", object:GetFullName(), object.ClassName)) end
		end
	end
	if count == 0 then return "No client-visible remotes were found in the selected replicated container." end
	local suffix = count > 80 and string.format("\n- ... %d additional remotes omitted", count - 80) or ""
	return table.concat(records, "\n") .. suffix
end

function Splash:_makeSteam(parent: Instance, x: number, delay: number)
	local steam = create("Frame", {
		Name = "Steam",
		AnchorPoint = Vector2.new(0.5, 1),
		Position = UDim2.new(0, x, 0, 70),
		Size = UDim2.fromOffset(4, 30),
		BackgroundColor3 = Color3.fromRGB(238, 222, 255),
		BackgroundTransparency = 0.25,
		Rotation = -8,
	}, parent)
	addCorner(steam, 4)

	task.delay(delay, function()
		if self.Destroyed or not steam.Parent then return end
		while not self.Destroyed and steam.Parent do
			steam.Position = UDim2.new(0, x, 0, 70)
			steam.Size = UDim2.fromOffset(4, 26)
			steam.BackgroundTransparency = 0.25
			steam.Rotation = -8
			tween(steam, TweenInfo.new(1.35, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), {
				Position = UDim2.new(0, x + 7, 0, 8),
				Size = UDim2.fromOffset(3, 38),
				BackgroundTransparency = 1,
				Rotation = 12,
			})
			task.wait(1.1)
		end
	end)
end

function Splash:_setProgress(value: number, status: string)
	self.ProgressValue = math.clamp(value, 0, 1)
	if self.ProgressFill then
		tween(self.ProgressFill, TweenInfo.new(0.28, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
			Size = UDim2.new(self.ProgressValue, 0, 1, 0),

		self.StatusLabel.Text = status
	end

function Splash:Run(stages: {{Name: string, Duration: number?}}?, onComplete: (() -> ())?)
	if not self.Gui then self:Build() end
	local selectedStages = stages or {
		{Name = "Creating interface...", Duration = 0.35},
		{Name = "Loading theme engine...", Duration = 0.3},
		{Name = "Registering components...", Duration = 0.35},
		{Name = "Preparing configuration...", Duration = 0.3},
		{Name = "Almost ready...", Duration = 0.25},
	}

	local total = #selectedStages
	for index, stage in ipairs(selectedStages) do
		if self.Destroyed then return end
		self:_setProgress((index - 1) / total, stage.Name)
		task.wait(stage.Duration or 0.25)
		self:_setProgress(index / total, stage.Name)
	end

	self:_setProgress(1, "Ready")
	task.wait(0.22)
	if onComplete then onComplete() end
end

function Splash:Close()
	if self.Destroyed then return end
	self.Destroyed = true
	if self.Gui then
		local background = self.Gui:FindFirstChild("Background")
		if background then
			tween(background, TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
				BackgroundTransparency = 1,
			})
			task.delay(0.38, function()
				if self.Gui then self.Gui:Destroy() end
			end)
		else
			self.Gui:Destroy()
		end
	end
end

function Splash:Destroy()
	self:Close()
end

	return Splash
end

function Library:_startTamperMonitor()
	-- Best-effort integrity check for accidental edits/removal in PlayerGui.
	-- This is not a security boundary: a client can disable or modify client code.
	if not self.Gui then return end
	self.Gui:SetAttribute("CozyIntegrity", "CozyV3")
	local expectedName = self.Gui.Name
	task.spawn(function()
		while self.Gui and not self.Destroyed do
			local reason: string? = nil
			if self.Gui.Parent ~= self.ExpectedParent then reason = "ScreenGui reparented" end
			if self.Gui.Name ~= expectedName then reason = "ScreenGui name changed" end
			if self.Gui:GetAttribute("CozyIntegrity") ~= "CozyV3" then reason = "integrity marker changed" end
			if not self.Gui:FindFirstChild("Root") then reason = "Root UI removed" end
			if not self.Gui:FindFirstChild("TabList", true) then reason = "Tab rail removed" end
			if reason then
				if self.TamperCallback then self.TamperCallback(reason) end
				self.Gui:SetAttribute("CozyIntegrity", "CozyV3")
				if self.Gui.Parent ~= self.ExpectedParent and self.ExpectedParent then self.Gui.Parent = self.ExpectedParent end
				task.wait(1)
			end
			task.wait(1)
		end
	end)
end

function Library:_setupDrag(handle: GuiObject)
	local dragging = false
	local dragStart = Vector2.zero
	local startPosition = UDim2.fromScale(0.5, 0.5)
	handle.Active = true
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPosition = self.Root.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging or (input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch) then return end
		local delta = input.Position - dragStart
		self.Root.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
	end)
end
function Library:_setRailExpanded(expanded: boolean)
	if self.IsTouchDevice then expanded = true end
	self.SideRailExpanded = expanded
	if not self.TabList then return end
	local width = expanded and 170 or 48
	local duration = self.SideRailAnimation and 0.22 or 0
	tween(self.TabList, TweenInfo.new(duration, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = UDim2.new(0, width, 1, -84)})
	for _, tab in pairs(self.Tabs) do tab.Button.Text = expanded and tab.FullText or tab.Icon end
end
function Library:SetSideRailAnimation(enabled: boolean)
	self.SideRailAnimation = enabled
	if self.IsTouchDevice then self:_setRailExpanded(true) else self:_setRailExpanded(not enabled) end
end
function Library:_setupSideRail()
	if not self.TabList then return end
	if self.IsTouchDevice then self:_setRailExpanded(true); return end
	self.TabList.MouseEnter:Connect(function() self:_setRailExpanded(true) end)
	self.TabList.MouseLeave:Connect(function() if self.SideRailAnimation then self:_setRailExpanded(false) end end)
	self:_setRailExpanded(false)
end

function Library:AddTab(tab: {Name: string, Icon: string?, Bottom: boolean?, Disabled: boolean?, Description: string?})
	assert(tab.Name and tab.Name ~= "", "Cozy:AddTab requires a Name")
	if self.Tabs[tab.Name] then return self.Tabs[tab.Name] end

	local page = create("ScrollingFrame", {
		Name = tab.Name:gsub("%W", ""),
		Size = UDim2.fromScale(1, 1),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		ScrollBarThickness = 3,
		ScrollBarImageColor3 = self.Accent,
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
		CanvasSize = UDim2.new(),
		Visible = false,
	}, self.PageHolder)
	create("TextLabel", {
		Name = "Heading",
		Size = UDim2.new(1, 0, 0, 32),
		BackgroundTransparency = 1,
		Text = tab.Name,
		Font = Enum.Font.GothamBold,
		TextSize = 24,
		TextColor3 = Color3.fromRGB(255, 255, 255),
		TextXAlignment = Enum.TextXAlignment.Left,
	}, page)
	if tab.Description then
		create("TextLabel", {
			Position = UDim2.fromOffset(0, 38),
			Size = UDim2.new(1, 0, 0, 28),
			BackgroundTransparency = 1,
			Text = tab.Description,
			Font = Enum.Font.Gotham,
			TextSize = 13,
			TextColor3 = Color3.fromRGB(170, 160, 184),
			TextXAlignment = Enum.TextXAlignment.Left,
		}, page)
	end

	local fullText = (tab.Icon or "") .. "  " .. tab.Name
	local button = create("TextButton", {
		Name = tab.Name .. "Button",
		Size = UDim2.new(1, -6, 0, 36),
		BackgroundColor3 = Color3.fromRGB(29, 22, 42),
		BackgroundTransparency = tab.Disabled and 0.35 or 0,
		Text = fullText,
		Font = Enum.Font.GothamMedium,
		TextSize = 13,
		TextColor3 = tab.Disabled and Color3.fromRGB(125, 116, 136) or Color3.fromRGB(224, 216, 232),
		AutoButtonColor = not tab.Disabled,
		LayoutOrder = tab.Bottom and 1000 or #self.Tabs,
	}, self.TabList)
	round(button, 8)
	create("UIGradient", {Color = ColorSequence.new(Color3.fromRGB(42, 30, 58), Color3.fromRGB(24, 18, 36)), Rotation = 90}, button)
	create("UIStroke", {Color = self.Accent, Transparency = 0.92, Thickness = 1}, button)
	self:_bindButtonMotion(button)

	if not tab.Disabled then
		button.Activated:Connect(function()
			self:_playButtonSound(tab.Name)
			self:SelectTab(tab.Name)
		end)
	end

	local record = {Name = tab.Name, Page = page, Button = button, Disabled = tab.Disabled, FullText = fullText, Icon = tab.Icon or ""}
	self.Tabs[tab.Name] = record
	return record
end

function Library:SelectTab(name: string)
	local selected = self.Tabs[name]
	if not selected or selected.Disabled then return end
	for _, tab in pairs(self.Tabs) do
		if tab ~= selected then
			tab.Page.Visible = false
			tween(tab.Button, TweenInfo.new(0.16, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {BackgroundColor3 = Color3.fromRGB(29, 22, 42)})
			local tabScale = tab.Button:FindFirstChildOfClass("UIScale")
			if tabScale then tween(tabScale, TweenInfo.new(0.16, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Scale = 1}) end
		end
	end
	selected.Page.Visible = true
	selected.Page.Position = UDim2.fromOffset(16, 0)
	tween(selected.Page, TweenInfo.new(0.28, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Position = UDim2.fromOffset(0, 0)})
	tween(selected.Button, TweenInfo.new(0.22, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {BackgroundColor3 = self.Accent})
	local selectedScale = selected.Button:FindFirstChildOfClass("UIScale")
	if selectedScale then tween(selectedScale, TweenInfo.new(0.22, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1.035}) end
	self.ActiveTab = name
end

function Library:SetGraphicsMode(mode: string)
	mode = string.upper(mode:sub(1, 1)) .. string.lower(mode:sub(2))
	if mode ~= "Low" and mode ~= "Medium" and mode ~= "High" then mode = "Medium" end
	self.GraphicsMode = mode
	local profile3D = self.Gui and self.Gui:FindFirstChild("Profile3D", true)
	if profile3D and profile3D:IsA("ViewportFrame") then profile3D.Visible = mode ~= "Low" and self.Model3DEnabled end
	local profile2D = self.Gui and self.Gui:FindFirstChild("Profile2D", true)
	if profile2D and profile2D:IsA("Frame") then profile2D.Visible = mode == "Low" or not self.Model3DEnabled end
	if mode == "Low" then
		self:SetSideRailAnimation(false)
	elseif mode == "High" then
		self:SetSideRailAnimation(true)
	end
	if self.Root then
		local stroke = self.Root:FindFirstChildOfClass("UIStroke")
		if stroke then stroke.Transparency = mode == "Low" and 0.88 or (mode == "High" and 0.55 or 0.72) end
	end
end

function Library:_bindButtonMotion(button: GuiButton)

	button.MouseLeave:Connect(function() tween(scale, TweenInfo.new(0.16, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Scale = 1}) end)
	button.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then tween(scale, TweenInfo.new(0.06, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Scale = 0.965}) end

	button.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then tween(scale, TweenInfo.new(0.12, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1.025}) end
	end)

local function cozyHttp(url: string, method: string, body: any?, headers: {[string]: string}?)
	local bridge = shared.CozyHttpRequest
	if type(bridge) ~= "function" then return nil, "Approved client HTTP bridge is not configured." end
	local encoded = body and HttpService:JSONEncode(body) or ""
	local request = {Url = url, Method = method, Headers = headers or { ["Content-Type"] = "application/json" }, Body = encoded}
	local ok, response = pcall(bridge, request)
	if not ok or type(response) ~= "table" then
		ok, response = pcall(bridge, url, method, headers or { ["Content-Type"] = "application/json" }, encoded)
	end
	if not ok or type(response) ~= "table" then return nil, "Client HTTP bridge failed." end
	local status = tonumber(response.StatusCode or response.Status or 0) or 0
	local raw = tostring(response.Body or response.body or "")
	local decoded = nil
	if raw ~= "" then pcall(function() decoded = HttpService:JSONDecode(raw) end) end
	if status < 200 or status >= 300 then return decoded, "HTTP " .. tostring(status) end
	return decoded, nil
end
local function fetchBackendSource()
	local bridge = shared.CozyHttpRequest
	if type(bridge) ~= "function" then return nil, "Approved client HTTP bridge is not configured." end
	local request = {Url = COZY_SOURCE_URL, Method = "GET", Headers = { ["x-cozy-ui-id"] = COZY_UI_ID, ["x-cozy-access-key"] = COZY_ACCESS_KEY }, Body = ""}
	local ok, response = pcall(bridge, request)
	if not ok or type(response) ~= "table" then
		ok, response = pcall(bridge, COZY_SOURCE_URL, "GET", request.Headers, "")
	end
	if not ok or type(response) ~= "table" then return nil, "Client HTTP bridge failed." end
	local status = tonumber(response.StatusCode or response.Status or 0) or 0
	local raw = tostring(response.Body or response.body or "")
	if status < 200 or status >= 300 then return nil, "HTTP " .. tostring(status) end
	return raw, nil
end
shared.CozyFetchSource = fetchBackendSource
local function requestCozyKey(action: string, key: string?)
	local endpoint = COZY_CUSTOM_KEY_URL ~= "" and COZY_CUSTOM_KEY_URL or (COZY_BACKEND_URL .. "/api/roblox/key/" .. action)
	return cozyHttp(endpoint, "POST", {uiId = COZY_UI_ID, key = key or ""}, { ["Content-Type"] = "application/json", ["x-cozy-ui-id"] = COZY_UI_ID })
end
local function requestCozyAI(prompt: string)
	local response, err = cozyHttp(COZY_BACKEND_URL .. "/api/roblox/ai", "POST", {prompt = prompt}, { ["Content-Type"] = "application/json", ["x-cozy-ui-id"] = COZY_UI_ID, ["x-cozy-access-key"] = COZY_ACCESS_KEY })
	if not response then return nil, err end
	return response.answer or response, nil
end
